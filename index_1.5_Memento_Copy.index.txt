(function buildAlphaMenu(){
  const menu = document.getElementById('alphaMenu');
  if (!menu) return;

  // Pull tools from the page
  const items = Array.from(document.querySelectorAll('.tool'))
    .map(tool => ({
      name: (tool.querySelector('.link')?.textContent || '').trim(),
      url: tool.querySelector('.url-link')?.getAttribute('href') || ''
    }))
    .filter(x => x.name && x.url);

  // Sort alphabetically by display name
  items.sort((a,b) => a.name.localeCompare(b.name, undefined, { sensitivity: 'base' }));

  // Group into A–Z + #
  const groups = new Map();
  const getKey = (s) => {
    const ch = (s || '').trim().charAt(0).toUpperCase();
    return (ch >= 'A' && ch <= 'Z') ? ch : '#';
  };

  items.forEach(it => {
    const k = getKey(it.name);
    if (!groups.has(k)) groups.set(k, []);
    groups.get(k).push(it);
  });

  // Clear any existing (keep first placeholder)
  menu.querySelectorAll('optgroup, option:not(:first-child)').forEach(n => n.remove());

  // Render optgroups in order
  const keys = Array.from(groups.keys()).sort((a,b) => (a === '#') - (b === '#') || a.localeCompare(b));
  keys.forEach(k => {
    const og = document.createElement('optgroup');
    og.label = k;
    groups.get(k).forEach(it => {
      const opt = document.createElement('option');
      opt.value = it.url;
      opt.textContent = it.name;
      og.appendChild(opt);
    });
    menu.appendChild(og);
  });

  // Open selected tool in a new tab
  menu.addEventListener('change', () => {
    const url = menu.value;
    if (!url) return;
    window.open(url, '_blank', 'noopener,noreferrer');
    menu.value = '';
  });
})();

    company: ["company", "corporate", "registr", "sec", "edgar", "payroll", "bankruptcy"],
    archives: ["archive", "cached", "wayback"],
    malware: ["malware", "sandbox", "threat", "intel", "ioc", "virus"],
    media: ["image", "video", "pdf", "metadata", "forensic", "exif"],
  };
  const keys = heuristics[wf] || [];
  return keys.some(k => hay.includes(k));
}

function toolMatchesChips(tool) {
  for (const chip of state.chips) {
    if (chip === "favorites") {
      if (!state.favorites.has(tool.id)) return false;
    }
    if (chip === "free") {
      if (tool.cost && tool.cost !== "free") return false;
      // If no cost metadata, don't exclude
    }
    if (chip === "api") {
      if (tool.api === false) return false;
      if (tool.api == null) return false; // strict: only show if explicitly api:true
    }
    if (chip === "account") {
      if (tool.requiresAccount === true) return false;
      if (tool.requiresAccount == null) return false; // strict: only show if explicitly false
    }
    if (chip === "gov") {
      const isGov = (tool.tags || []).includes("gov") || normalize(tool.url).includes(".gov");
      if (!isGov) return false;
    }
  }
  return true;
}

function getFilteredTools() {
  const { tools } = state.data;

  let results = tools;

  // Fuzzy search (if query)
  const q = normalize(state.query);
  if (q) {
    const fuseResults = state.fuse.search(q, { limit: 5000 });
    results = fuseResults.map(r => r.item);
  }

  // Workflow + chips filters
  results = results.filter(t => toolMatchesWorkflow(t, state.workflow)).filter(toolMatchesChips);

  // Sorting
  if (state.sort === "name") {
    results = [...results].sort((a, b) => a.name.localeCompare(b.name));
  } else if (state.sort === "category") {
    results = [...results].sort((a, b) => (a.categoryId || "").localeCompare(b.categoryId || "") || a.name.localeCompare(b.name));
  } else if (state.sort === "fav") {
    results = [...results].sort((a, b) => {
      const af = state.favorites.has(a.id) ? 1 : 0;
      const bf = state.favorites.has(b.id) ? 1 : 0;
      return bf - af || a.name.localeCompare(b.name);
    });
  }

  return results;
}

function updateProgressBar() {
  const scroll = window.scrollY;
  const height = document.body.scrollHeight - window.innerHeight;
  const p = height <= 0 ? 0 : (scroll / height) * 100;
  $("#progress").style.width = `${Math.min(100, Math.max(0, p))}%`;
}

function downloadText(filename, text, type = "text/plain") {
  const blob = new Blob([text], { type });
  const a = document.createElement("a");
  a.href = URL.createObjectURL(blob);
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  setTimeout(() => {
    URL.revokeObjectURL(a.href);
    a.remove();
  }, 0);
}

function exportJSON(tools) {
  const payload = {
    meta: { exportedAt: new Date().toISOString(), count: tools.length },
    tools,
  };
  downloadText("fireview-export.json", JSON.stringify(payload, null, 2), "application/json");
}

function exportCSV(tools) {
  const header = ["name", "url", "categoryId", "description", "tags", "cost", "requiresAccount", "api", "region", "workflow"];
  const rows = tools.map(t => [
    t.name,
    t.url,
    t.categoryId || "",
    (t.description || "").replaceAll("\n", " "),
    Array.isArray(t.tags) ? t.tags.join("|") : "",
    t.cost || "",
    t.requiresAccount ?? "",
    t.api ?? "",
    t.region || "",
    Array.isArray(t.workflow) ? t.workflow.join("|") : "",
  ]);
  const csv = [header, ...rows]
    .map(r => r.map(v => `"${String(v ?? "").replaceAll(`"`, `""`)}"`).join(","))
    .join("\n");
  downloadText("fireview-export.csv", csv, "text/csv");
}

function exportMarkdown(tools, categoriesById) {
  const byCat = new Map();
  for (const t of tools) {
    const key = t.categoryId || "uncategorized";
    if (!byCat.has(key)) byCat.set(key, []);
    byCat.get(key).push(t);
  }

  let md = `# Fireview Tool Directory (Export)\n\nExported: ${new Date().toISOString()}\nCount: ${tools.length}\n\n`;
  for (const [catId, list] of byCat.entries()) {
    const catName = categoriesById.get(catId)?.name || catId;
    md += `## ${catName}\n\n`;
    for (const t of list.sort((a, b) => a.name.localeCompare(b.name))) {
      md += `- **${t.name}** — ${t.url}\n`;
      if (t.description) md += `  - ${t.description}\n`;
      if (t.tags?.length) md += `  - Tags: ${t.tags.join(", ")}\n`;
    }
    md += `\n`;
  }
  downloadText("fireview-export.md", md, "text/markdown");
}

async function copyToClipboard(text) {
  try {
    await navigator.clipboard.writeText(text);
    return true;
  } catch {
    return false;
  }
}

function render() {
  const main = $("#main");
  main.innerHTML = "";

  const categoriesById = new Map(state.data.categories.map(c => [c.id, c]));
  const filtered = getFilteredTools();

  // Dynamic counts per category
  const countByCat = new Map();
  for (const t of filtered) {
    countByCat.set(t.categoryId, (countByCat.get(t.categoryId) || 0) + 1);
  }

  // Category nav
  const nav = $("#catnav");
  nav.innerHTML = "";
  for (const c of state.data.categories) {
    const count = countByCat.get(c.id) || 0;
    const a = document.createElement("a");
    a.className = "catlink";
    a.href = `#${c.id}`;
    a.innerHTML = `${c.name} <span class="count">(${count})</span>`;
    nav.appendChild(a);
  }

  // Status line
  $("#status").textContent = `${filtered.length} tools shown • ${state.data.tools.length} total`;

  // Render categories in order, but only those with tools in view
  for (const c of state.data.categories) {
    const toolsInCat = filtered.filter(t => t.categoryId === c.id);
    if (!toolsInCat.length) continue;

    const section = document.createElement("section");
    section.className = "category";
    section.id = c.id;

    const head = document.createElement("div");
    head.className = "cathead";

    const h2 = document.createElement("h2");
    h2.textContent = c.name;

    const meta = document.createElement("div");
    meta.className = "catmeta";

    const count = document.createElement("div");
    count.className = "muted";
    count.textContent = `${toolsInCat.length} tools`;

    const btnOpenAll = document.createElement("button");
    btnOpenAll.className = "btn small ghost";
    btnOpenAll.textContent = "Open all";
    btnOpenAll.title = "Open all tools in this category";
    btnOpenAll.addEventListener("click", () => {
      const ok = confirm(`Open ${toolsInCat.length} tabs? Your browser may block popups.`);
      if (!ok) return;
      for (const t of toolsInCat) window.open(t.url, "_blank", "noopener,noreferrer");
    });

    meta.append(count, btnOpenAll);
    head.append(h2, meta);

    const ul = document.createElement("ul");
    ul.className = "grid";

    for (const t of toolsInCat) {
      ul.appendChild(renderCard(t));
    }

    section.append(head, ul);
    main.appendChild(section);
  }

  // If nothing matched
  if (!main.children.length) {
    const empty = document.createElement("div");
    empty.className = "category";
    empty.innerHTML = `<h2>No results</h2><div class="muted">Try a different search or clear filters.</div>`;
    main.appendChild(empty);
  }

  // Export buttons
  $("#btnExportJson").onclick = () => exportJSON(filtered);
  $("#btnExportCsv").onclick = () => exportCSV(filtered);
  $("#btnExportMd").onclick = () => exportMarkdown(filtered, categoriesById);
}

function renderCard(t) {
  const li = document.createElement("li");
  li.className = "card";

  const top = document.createElement("div");
  top.className = "cardtop";

  const left = document.createElement("div");
  left.style.flex = "1";

  const name = document.createElement("h3");
  name.className = "name";
  name.textContent = t.name;

  const desc = document.createElement("p");
  desc.className = "desc";
  desc.textContent = t.description || (t.keywords ? `Keywords: ${t.keywords.slice(0, 90)}${t.keywords.length > 90 ? "…" : ""}` : "—");

  const url = document.createElement("a");
  url.className = "url";
  url.href = t.url;
  url.target = "_blank";
  url.rel = "noopener noreferrer";
  url.textContent = t.url;

  const favBtn = document.createElement("button");
  favBtn.className = "iconbtn";
  favBtn.textContent = state.favorites.has(t.id) ? "⭐ Favorited" : "☆ Favorite";
  if (state.favorites.has(t.id)) favBtn.classList.add("active");
  favBtn.addEventListener("click", () => {
    if (state.favorites.has(t.id)) state.favorites.delete(t.id);
    else state.favorites.add(t.id);
    saveFavorites();
    render();
  });

  left.append(name, desc, url);
  top.append(left, favBtn);

  const pills = document.createElement("div");
  pills.className = "pills";

  // Pill helpers (only show if present)
  const addPill = (text, cls = "") => {
    const s = document.createElement("span");
    s.className = `pill ${cls}`.trim();
    s.textContent = text;
    pills.appendChild(s);
  };

  if (t.cost) addPill(t.cost, t.cost === "free" ? "good" : "");
  if (t.api === true) addPill("API", "good");
  if (t.requiresAccount === false) addPill("No account", "good");
  if (t.region) addPill(t.region.toUpperCase());
  if (Array.isArray(t.tags)) t.tags.slice(0, 4).forEach(tag => addPill(tag));

  const actions = document.createElement("div");
  actions.className = "cardactions";

  const leftActions = document.createElement("div");
  leftActions.className = "leftactions";

  const btnOpen = document.createElement("button");
  btnOpen.className = "iconbtn";
  btnOpen.textContent = "Open";
  btnOpen.addEventListener("click", () => window.open(t.url, "_blank", "noopener,noreferrer"));

  const btnCopy = document.createElement("button");
  btnCopy.className = "iconbtn";
  btnCopy.textContent = "Copy URL";
  btnCopy.addEventListener("click", async () => {
    const ok = await copyToClipboard(t.url);
    btnCopy.textContent = ok ? "Copied ✔️" : "Copy failed";
    setTimeout(() => (btnCopy.textContent = "Copy URL"), 900);
  });

  const btnNote = document.createElement("button");
  btnNote.className = "iconbtn";
  btnNote.textContent = "📝 Note";

  leftActions.append(btnOpen, btnCopy, btnNote);

  const rightActions = document.createElement("div");
  rightActions.className = "rightactions";

  const note = document.createElement("textarea");
  note.className = "notearea";
  note.rows = 2;
  note.placeholder = "Personal note (saved locally)…";
  note.value = notesMap.get(t.id) || "";
  note.addEventListener("input", () => {
    const v = note.value.trim();
    if (v) notesMap.set(t.id, v);
    else notesMap.delete(t.id);
    saveNotes();
  });

  btnNote.addEventListener("click", () => {
    note.classList.toggle("open");
    if (note.classList.contains("open")) note.focus();
  });

  actions.append(leftActions, rightActions);

  li.append(top);
  if (pills.children.length) li.append(pills);
  li.append(actions, note);

  return li;
}

function wireUI() {
  const q = $("#q");
  const btnClear = $("#btnClear");

  q.addEventListener("input", () => {
    state.query = q.value;
    render();
  });

  btnClear.addEventListener("click", () => {
    q.value = "";
    state.query = "";
    render();
    q.focus();
  });

  $("#workflow").addEventListener("change", (e) => {
    state.workflow = e.target.value;
    render();
  });

  $("#sort").addEventListener("change", (e) => {
    state.sort = e.target.value;
    render();
  });

  // Chips
  $$(".chip").forEach(chip => {
    chip.addEventListener("click", () => {
      const f = chip.dataset.filter;
      if (state.chips.has(f)) {
        state.chips.delete(f);
        chip.classList.remove("active");
      } else {
        state.chips.add(f);
        chip.classList.add("active");
      }
      render();
    });
  });

  // Theme
  $("#btnTheme").addEventListener("click", () => {
    setTheme(state.theme === "light" ? "dark" : "light");
  });

  // Help
  const helpDialog = $("#helpDialog");
  $("#btnHelp").addEventListener("click", () => helpDialog.showModal());

  // Keyboard shortcuts
  document.addEventListener("keydown", (e) => {
    if (e.key === "/" && document.activeElement !== q) {
      e.preventDefault();
      q.focus();
      return;
    }
    if (e.key === "Escape") {
      if (helpDialog.open) helpDialog.close();
      q.value = "";
      state.query = "";
      render();
      return;
    }
    if (e.key.toLowerCase() === "f" && !e.ctrlKey && !e.metaKey && !e.altKey) {
      // toggle favorites chip
      const chip = $(`.chip[data-filter="favorites"]`);
      chip?.click();
      return;
    }
    if (e.key.toLowerCase() === "t") {
      $("#btnTheme").click();
      return;
    }
    if (e.key === "?") {
      helpDialog.showModal();
      return;
    }
  });

  // Scroll progress
  window.addEventListener("scroll", updateProgressBar, { passive: true });
  updateProgressBar();
}

async function init() {
  setTheme(state.theme);

  const res = await fetch("./tools.json", { cache: "no-store" });
  if (!res.ok) throw new Error(`Failed to load tools.json (${res.status})`);
  state.data = await res.json();

  $("#siteTitle").textContent = state.data?.meta?.title || "Fireview Tool Directory";

  // Setup Fuse
  state.fuse = new Fuse(state.data.tools, {
    includeScore: true,
    threshold: 0.35,
    ignoreLocation: true,
    keys: [
      { name: "name", weight: 0.5 },
      { name: "description", weight: 0.25 },
      { name: "keywords", weight: 0.2 },
      { name: "url", weight: 0.05 },
      { name: "tags", weight: 0.2 },
    ],
  });

  wireUI();
  render();
}

init().catch(err => {
  console.error(err);
  $("#status").textContent = `Error: ${err.message}`;
});
